namespace = hmm_judaism_events

# A Jewish revival in Europe
country_event = {
	id = hmm_judaism_events.1
	title = hmm_judaism_events.1.t
	desc = hmm_judaism_events.1.d
	picture = JUDAISM_eventPicture

	major = yes
	fire_only_once = yes

	trigger = {
		ai = no
		capital_scope = {
			continent = europe
		}
		current_age = age_of_reformation
		religion = catholic
		is_defender_of_faith = no
		is_religion_enabled = protestant
        NOT = { has_country_flag = had_jewish_question }
	}

	mean_time_to_happen = {
		months = 12
	}

	immediate = {
		set_country_flag = had_jewish_question
		if = {
			limit = {
				capital_scope = {
					owned_by = ROOT
					controlled_by = ROOT
				}
			}
			capital_scope = { save_event_target_as = jewish_revival_province }
		}
		else = {
			random_owned_province = {
				save_event_target_as = jewish_revival_province
			}
		}
	}

	option = {
		name = hmm_judaism_events.1.a
		enable_religion = jewish
		change_religion = jewish
		event_target:jewish_revival_province = {
			change_religion = jewish
		}
		add_country_modifier = {
			name = "hmm_heathen_conversion_zeal"
			duration = 3650
		}
	}
	option = {
		name = hmm_judaism_events.1.b
        custom_tooltip = hmm_judaism_events_1b_tt
        hidden_effect = {
            every_owned_province = {
                add_province_modifier = {
                    name = "hmm_refused_jewish_question_modifier"
                    duration = 18250
                }
            }
        }
	}
}
