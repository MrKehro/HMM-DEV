# Restoration of the Ilkhanate (alt history)
# by Daniel Tolman
country_event = {
	id = flavor_cri.8
	title = "flavor_cri.8.t"
	desc = "flavor_cri.8.d"
	picture = MONGOL_BANNERS_eventPicture
	
	fire_only_once = yes
	is_triggered_only = yes
	major = yes
	
	option = {
		name = flavor_cri.8.a
		persia_region = {
			limit = {
				owned_by = ROOT
			}
			add_core = ILK
			if = {
				limit = { is_core = ROOT }
				remove_core = ROOT
			}
		}
		release = ILK
		create_march = ILK
		ILK = {
			change_government = monarchy
			#set_legacy_government = despotic_monarchy
			change_primary_culture = ROOT
			change_religion = ROOT
			define_ruler = {
				dynasty = ROOT
				culture = ROOT
				religion = ROOT
			}
		}
	}
	option = {
		name = flavor_cri.8.b
		4338 = {
			add_base_tax = 2
			add_base_manpower = 2
			add_base_production = 2
		}
	}
}
