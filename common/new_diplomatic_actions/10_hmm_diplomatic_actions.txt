# HMM custom diplomatic actions
# ROOT is the country using the action, FROM is the target country.


hmm_scorrupt_officials = {
	category = spy_offence

	require_acceptance = no

	is_visible = {
		full_idea_group = spy_ideas
	}
	is_allowed = {
		FROM = { NOT = { has_country_modifier = corrupt_officials } }
		custom_trigger_tooltip = {
			tooltip = REQUIRES_30_SPY_NETWORK_TT
			has_spy_network_in = {
				who = FROM
				value = 30
			}
		}
	}
	on_accept = {
		add_spy_network_in = {
			who = FROM
			value = -30
		}
        FROM = {
            add_country_modifier = {
                name = corrupt_officials
                duration = 1825
            }
        }
	}
	on_decline = {
	}

	ai_will_do = { # this is a trigger
		has_spy_network_in = {
			who = FROM
			value = 30
		}
	}
}

hmm_agitate_for_liberty = {
	category = spy_offence

	require_acceptance = no

	is_visible = {
		full_idea_group = spy_ideas
		FROM = { is_subject = yes }
	}
	is_allowed = {
		FROM = { NOT = { has_country_modifier = hmm_agitated_for_liberty_desire } }
		custom_trigger_tooltip = {
			tooltip = REQUIRES_40_SPY_NETWORK_TT
			FROM = {
				overlord = {
					ROOT = {
						has_spy_network_in = {
							who = PREV
							value = 40
						}
					}
				}
			}
		}
	}
	on_accept = {
		FROM = {
			overlord = {
				ROOT = {
					add_spy_network_in = {
						who = PREV
						value = -40
					}
				}
			}
		}
		FROM = {
			add_country_modifier = {
                name = hmm_agitated_for_liberty_desire
                duration = 1825
            }
		}
	}
	on_decline = {
	}

	ai_will_do = {
		has_spy_network_in = {
			who = FROM
			value = 90
		}
	}
}
